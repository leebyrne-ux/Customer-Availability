Google Sheet Web App Connection

Important: Make sure your Apps Script Web App deployment has Who has access set to Anyone so GitHub Pages can read and write rows without a login block. Also ensure your Code.gs matches the 11-column script below: 
function doGet(e) {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  const rows = sheet.getDataRange().getValues();
  if (rows.length <= 1) return ContentService.createTextOutput(JSON.stringify([])).setMimeType(ContentService.MimeType.JSON);
  const data = rows.slice(1).map(r => ({
    id: r[0],
    offerDate: r[1],
    customer: r[2],
    origin: r[3],
    destination: r[4],
    collectionDate: r[5],
    deliveryDate: r[6],
    isHaz: r[7] === 'YES' || r[7] === true,
    isLtdQty: r[8] === 'YES' || r[8] === true,
    status: r[9] || 'Available',
    region: r[10] || ''
  }));
  return ContentService.createTextOutput(JSON.stringify(data)).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(['ID', 'Offer Date', 'Customer', 'Loading From', 'Delivery To', 'Collection Date', 'Delivery Date', 'HAZ', 'Ltd Qty', 'Status', 'Region']);
  }
  const payload = JSON.parse(e.postData.contents);
  if (payload.action === 'add') {
    const items = Array.isArray(payload.loads) ? payload.loads : [payload.load];
    items.forEach(l => {
      sheet.appendRow([l.id, l.offerDate, l.customer, l.origin, l.destination, l.collectionDate, l.deliveryDate, l.isHaz ? 'YES' : '', l.isLtdQty ? 'YES' : '', l.status, l.region || '']);
    });
  } else if (payload.action === 'updateStatus') {
    const rows = sheet.getDataRange().getValues();
    for (let i = 1; i < rows.length; i++) {
      if (String(rows[i][0]) === String(payload.id)) {
        sheet.getRange(i + 1, 10).setValue(payload.status);
        break;
      }
    }
  } else if (payload.action === 'delete') {
    const rows = sheet.getDataRange().getValues();
    for (let i = rows.length - 1; i >= 1; i--) {
      if (String(rows[i][0]) === String(payload.id)) {
        sheet.deleteRow(i + 1);
        break;
      }
    }
  }
  return ContentService.createTextOutput(JSON.stringify({ ok: true })).setMimeType(ContentService.MimeType.JSON);
}
